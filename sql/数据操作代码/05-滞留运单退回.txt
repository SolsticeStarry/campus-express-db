SET @r = (SELECT parcel_no FROM parcel
          WHERE status = '滞留' AND slot_id IS NOT NULL
          ORDER BY parcel_no LIMIT 1);

UPDATE parcel
SET status = '退回'
WHERE parcel_no = @r;

SELECT p.parcel_no AS 运单号,
       p.status AS 运单状态,
       sl.slot_code AS 柜格口,
       sl.status AS 柜格状态
FROM parcel p
JOIN locker_slot sl ON sl.slot_id = p.slot_id
WHERE p.parcel_no = @r;
