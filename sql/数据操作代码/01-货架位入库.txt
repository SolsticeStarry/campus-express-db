SELECT
    p.parcel_no AS 运单号,
    st.station_name AS 驿站,
    sh.shelf_code AS 货架位,
    sh.current_count AS 当前数量,
    sh.capacity AS 容量,
    sh.status AS 状态
FROM parcel p
JOIN shelf sh ON sh.shelf_id = p.shelf_id
JOIN station st ON sh.station_id = st.station_id
WHERE p.parcel_no = 'DEMO-SHELF-0001';
